cmake_minimum_required(VERSION 3.21)
project(ps3dmv_experiments VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(OpenMP QUIET)
add_library(holo_common src/io.cpp src/propagation.cpp)
target_include_directories(holo_common PUBLIC include)
target_include_directories(holo_common SYSTEM PUBLIC third_party)
target_link_libraries(holo_common PUBLIC Threads::Threads)
target_compile_definitions(holo_common PRIVATE POCKETFFT_CACHE_SIZE=16)
if(OpenMP_CXX_FOUND)
  target_link_libraries(holo_common PUBLIC OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_definitions(holo_common PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(holo_common PUBLIC /utf-8 /W4 /permissive- /EHsc)
else()
  target_compile_options(holo_common PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(reconstruct_original src/reconstruct_original.cpp)
target_link_libraries(reconstruct_original PRIVATE holo_common)
add_executable(reconstruct_original_fast src/reconstruct_original.cpp)
target_link_libraries(reconstruct_original_fast PRIVATE holo_common)
add_executable(generate_sintel_holograms src/generate_sintel_holograms.cpp)
target_link_libraries(generate_sintel_holograms PRIVATE holo_common)
add_executable(reconstruct_holo_preview src/reconstruct_holo_preview.cpp)
target_link_libraries(reconstruct_holo_preview PRIVATE holo_common)
enable_testing()
add_executable(holo_tests tests/test_core.cpp)
target_link_libraries(holo_tests PRIVATE holo_common)
add_test(NAME holo_core COMMAND holo_tests)
add_executable(frame_phase_tests tests/test_frame_phase.cpp)
target_link_libraries(frame_phase_tests PRIVATE holo_common)
add_test(NAME frame_phase COMMAND frame_phase_tests)
add_executable(ra_suite unified/ra_suite.cpp hm/study_metrics.cpp)
target_link_libraries(ra_suite PRIVATE holo_common bcrypt)
add_executable(ps_search_equivalence_test vvc444/ps_search_equivalence_test.cpp)
target_compile_options(ps_search_equivalence_test PRIVATE /utf-8 /W4 /EHsc /permissive-)
add_test(NAME ps_search_equivalence COMMAND ps_search_equivalence_test)
add_executable(study_io_test hm/study_io_test.cpp)
target_link_libraries(study_io_test PRIVATE holo_common bcrypt)
add_test(NAME study_io COMMAND study_io_test "${CMAKE_CURRENT_BINARY_DIR}/study_io_fixture")
add_executable(suite_scheduler_tests unified/scheduler_tests.cpp hm/study_metrics.cpp)
target_link_libraries(suite_scheduler_tests PRIVATE holo_common bcrypt)
add_test(NAME suite_scheduler COMMAND suite_scheduler_tests "${CMAKE_CURRENT_SOURCE_DIR}/../..")
